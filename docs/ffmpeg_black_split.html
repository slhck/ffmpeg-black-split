 65        Get black periods from ffmpeg.
 66

 69                                                  It must be a non-negative floating point number.
 70            picture_black_ratio_th (float, optional): Set the threshold for considering a picture 'black'.
 71            pixel_black_th (float, optional): Set the threshold for considering a pixel 'black'.
 72
 73        Returns:
 74            list: List of black periods.
 75        """
 76        black_periods: list[Period] = []
 77
 78        blackdetect_option_pairs = {
 79            "black_min_duration": black_min_duration,
 80            "picture_black_ratio_th": picture_black_ratio_th,
 81            "pixel_black_th": pixel_black_th,
 82        }
 83
 84        blackdetect_options = [
 85            f"{key}={value}" for key, value in blackdetect_option_pairs.items()
 86        ]
 87
 88        try:
 89            cmd = [
 90                self.ffmpeg_path,
 91                "-hide_banner",
 92                "-y",
 93                "-i",
 94                self.input_file,
 95                "-vf",
 96                f"blackdetect={':'.join(blackdetect_options)}",
 97                "-an",
 98                "-f",
 99                "null",
100                "-",
101            ]
102
103            cmd_q = " ".join([shlex.quote(c) for c in cmd])
104            logger.debug("Running ffmpeg command: {}".format(cmd_q))
105
106            ff = FfmpegProgress(cmd)
107            if self.progress:
108                with tqdm(total=100, position=1) as pbar:
109                    for p in ff.run_command_with_progress():
110                        pbar.update(p - pbar.n)
111            else:
112                for _ in ff.run_command_with_progress():
113                    pass
114
115            if ff.stderr is None:
116                raise Exception("No stderr from ffmpeg")
117
118            # Older ffmpeg (e.g. 7.1) prefixes the lines with "[blackdetect @ 0x...]",
119            # newer ffmpeg (e.g. 9.0) with the filter instance name, "[Parsed_blackdetect_0 @ 0x...]"
120            blackdetect_lines = [
121                line
122                for line in ff.stderr.splitlines()
123                if re.match(r"\[(?:Parsed_)?blackdetect(?:_\d+)? @ ", line)
124            ]
125
126            if len(blackdetect_lines) == 0:
127                print("No black periods detected.", file=sys.stderr)
128                return black_periods
129
130            for line in blackdetect_lines:
131                # [blackdetect @ 0x137f36f30] black_start:20 black_end:24.96 black_duration:4.96
132                #
133                # extract the black_start, black_end and black_duration values
134                black_start: Union[float, None] = None
135                black_end: Union[float, None] = None
136                black_duration: Union[float, None] = None
137                if black_start_match := re.search(r"black_start:(\d+(?:\.\d+)?)", line):
138                    black_start = float(black_start_match.group(1))
139                if black_end_match := re.search(r"black_end:(\d+(?:\.\d+)?)", line):
140                    black_end = float(black_end_match.group(1))
141                if black_duration_match := re.search(
142                    r"black_duration:(\d+(?:\.\d+)?)", line

144                    black_duration = float(black_duration_match.group(1))
145
146                if black_start is None or black_end is None or black_duration is None:
147                    raise Exception("Could not parse blackdetect line: {}".format(line))
148
149                black_periods.append(
150                    {
151                        "start": black_start,
152                        "end": black_end,
153                        "duration": black_duration,
154                    }
155                )

167    def black_periods_to_content_periods(
168        black_periods: list[Period],
169    ) -> list[Union[Period, OpenPeriod]]:
170        """
171        Calculate the inverted black periods to get the content periods.
172
173        Args:
174            black_periods (list): List of black periods.
175
176        Returns:
177            list: List of content periods.
178        """
179        content_periods: list[Union[Period, OpenPeriod]] = []
180        previous_period_end: float = 0.0
181
182        sorted_black_periods = sorted(black_periods, key=lambda x: x["start"])
183
184        for black_period in sorted_black_periods:
185            if black_period["start"] > previous_period_end:
186                content_periods.append(
187                    {"start": previous_period_end, "end": black_period["start"]}
188                )
189            previous_period_end = black_period["end"]
190
191        # add a final, open-ended one
192        content_periods.append({"start": previous_period_end, "end": None})
193
194        return content_periods

245    @staticmethod
246    def filter_black_periods(
247        black_periods: list[Period], num_cuts: int = 1
248    ) -> list[Period]:
249        """
250        Filter black periods based on desired number of cuts or custom black periods.
251
252        Args:
253            num_cuts (int): Desired number of cuts (default: 1)
254
255        Returns:
256            list[Period]: Filtered list of black periods
257        """
258        if not black_periods:
259            raise ValueError(
260                "No black periods detected. Run detect_black_periods first."
261            )
262
263        if num_cuts >= len(black_periods):
264            return black_periods
265
266        # select periods closest to evenly dividing the video
267        video_duration = max(period["end"] for period in black_periods)
268        ideal_cut_times = [
269            video_duration * (i + 1) / (num_cuts + 1) for i in range(num_cuts)
270        ]
271
272        selected_periods = []
273        for cut_time in ideal_cut_times:
274            closest_period = min(
275                black_periods,
276                key=lambda p: abs(p["start"] + p["duration"] / 2 - cut_time),
277            )
278            selected_periods.append(closest_period)
279            black_periods.remove(closest_period)
280
281        return selected_periods
